01024 
01025  int placa;
01026 
01027   //-- Si los 4 bits mas significativos de IPV estan a cero
01028   //-- es que se trata de una placa hecha por el usuario
01029   if ((ipv&0xF0) == 0) {
01030     strcpy(cad,PLACA_USER_CAD);
01031     return;
01032   }
01033 
01034   //-- El campo im junto con los 4 bits mas significativos de IPV
01035   //-- determinan el tipo de placa que se esta usando
01036   placa = ((im<<4)&0xFF0) | ((ipv>>4)&0x0F);
01037 
01038   switch(placa) {
01039   case PLACA_XPIC:
01040         strcpy(cad,PLACA_XPIC_CAD);
01041         break;
01042   case PLACA_CT6811:
