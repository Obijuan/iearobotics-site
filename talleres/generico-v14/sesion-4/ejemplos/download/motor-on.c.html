//-- Especificar el pic a emplear

#include <pic16f876a.h>

//-- Definiciones para los motores. Estos valores define
//-- los movimientos del robot

#define   AVANZA     0x1C
#define   ATRAS      0x16
#define   IZQUIERDA  0x1E
#define   DERECHA    0x14
#define   STOP       0x00

//----------------------------
//- Comienzo del programa  
//----------------------------

void main(void) {
        //-- Configurar el puerto B para trabajar con el Skybot
        //-- RB0, RB5, RB6 y RB7 como entrada
        //-- RB1, RB2, RB3 y RB4 como salidas
        TRISB=0xE1;
        
        //-- Activar los bits de los motores para que el robot avance
        PORTB=AVANZA;
        
        //-- Bucle infinito
        while(1);
}
