#include "libreria_skybot.h"


//----------------------------
//- Comienzo del programa  
//----------------------------

void main(void)
{
  ConfigurarSkybot();
  
  for (;;)
  {
    if (SensorIR_Izq_Lee_Negro() && SensorIR_Der_Lee_Negro())
    {
      Avanzar();
    }
    else if (SensorIR_Izq_Lee_Negro())
    {
      GirarIzq();
    }
    else if (SensorIR_Der_Lee_Negro())
    {
      GirarDer();
    }
    else
    {
      Parar();
    }
  }  
}
